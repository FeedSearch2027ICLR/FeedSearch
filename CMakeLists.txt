cmake_minimum_required(VERSION 3.18)
project(FeedSearch LANGUAGES CXX)

option(FEEDSEARCH_BUILD_ADAPTERS "Build HNSW, NSG and Vamana adapters" ON)
option(FEEDSEARCH_BUILD_BENCHMARKS "Build benchmark tools (does not run experiments)" ON)
option(FEEDSEARCH_BUILD_TESTS "Build small interface tests" ON)
option(FEEDSEARCH_NATIVE_ARCH "Optimize adapters for the build machine" ON)

add_library(feedsearch_core INTERFACE)
add_library(FeedSearch::core ALIAS feedsearch_core)
target_compile_features(feedsearch_core INTERFACE cxx_std_17)
target_include_directories(feedsearch_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)

if(FEEDSEARCH_BUILD_ADAPTERS)
  find_package(OpenMP REQUIRED)
  set(DISKANN_DIR "" CACHE PATH "DiskANN source or build tree")
  find_path(HNSWLIB_DIR hnswlib.h REQUIRED)
  find_path(FAISS_INCLUDE_DIR faiss/IndexNSG.h REQUIRED)
  find_library(FAISS_LIBRARY faiss REQUIRED)
  find_path(DISKANN_INCLUDE_DIR in_mem_data_store.h HINTS ${DISKANN_DIR}/include REQUIRED)
  find_library(DISKANN_LIBRARY diskann HINTS ${DISKANN_DIR}/build/src REQUIRED)
  set(DISKANN_EXTRA_LIBRARIES "aio;pthread;dl" CACHE STRING "Additional libraries required by the DiskANN build")
  set(ADAPTER_SOURCES src/backend.cpp src/hnsw_backend.cpp src/nsg_backend.cpp src/vamana_backend.cpp)

  function(configure_adapters target)
    target_link_libraries(${target} PUBLIC feedsearch_core OpenMP::OpenMP_CXX ${FAISS_LIBRARY} ${DISKANN_LIBRARY} openblas ${DISKANN_EXTRA_LIBRARIES})
    target_include_directories(${target} PRIVATE ${HNSWLIB_DIR} ${FAISS_INCLUDE_DIR} ${DISKANN_INCLUDE_DIR})
    target_compile_options(${target} PRIVATE -O3)
    if(FEEDSEARCH_NATIVE_ARCH)
      target_compile_options(${target} PRIVATE -march=native)
    endif()
  endfunction()

  add_library(feedsearch_adapters STATIC ${ADAPTER_SOURCES})
  add_library(FeedSearch::adapters ALIAS feedsearch_adapters)
  configure_adapters(feedsearch_adapters)

  if(FEEDSEARCH_BUILD_BENCHMARKS)
    add_library(feedsearch_adapters_count STATIC ${ADAPTER_SOURCES})
    configure_adapters(feedsearch_adapters_count)
    target_compile_definitions(feedsearch_adapters_count PUBLIC FEED_COUNT_DISTANCES=1)
    foreach(target feed_benchmark feed_count)
      add_executable(${target} benchmarks/benchmark.cpp)
      target_compile_options(${target} PRIVATE -O3)
      if(FEEDSEARCH_NATIVE_ARCH)
        target_compile_options(${target} PRIVATE -march=native)
      endif()
    endforeach()
    target_link_libraries(feed_benchmark PRIVATE feedsearch_adapters)
    target_link_libraries(feed_count PRIVATE feedsearch_adapters_count)
    add_executable(feed_build_index benchmarks/build_index.cpp)
    target_link_libraries(feed_build_index PRIVATE feedsearch_adapters)
    target_include_directories(feed_build_index PRIVATE ${HNSWLIB_DIR} ${FAISS_INCLUDE_DIR})
    target_compile_options(feed_build_index PRIVATE -O3)
    add_executable(feed_groundtruth benchmarks/groundtruth.cpp)
    target_link_libraries(feed_groundtruth PRIVATE feedsearch_core OpenMP::OpenMP_CXX openblas)
    find_path(OPENBLAS_CBLAS_DIR cblas.h PATHS /usr/include/x86_64-linux-gnu/openblas-pthread /usr/include/openblas NO_DEFAULT_PATH REQUIRED)
    target_include_directories(feed_groundtruth BEFORE PRIVATE ${OPENBLAS_CBLAS_DIR})
    target_compile_options(feed_groundtruth PRIVATE -O3)
    if(FEEDSEARCH_NATIVE_ARCH)
      target_compile_options(feed_build_index PRIVATE -march=native)
      target_compile_options(feed_groundtruth PRIVATE -march=native)
    endif()
  endif()
elseif(FEEDSEARCH_BUILD_BENCHMARKS)
  message(STATUS "Adapters disabled: skipping benchmark tools")
endif()

if(FEEDSEARCH_BUILD_TESTS)
  enable_testing()
  add_executable(feed_tests tests/test_interfaces.cpp)
  target_link_libraries(feed_tests PRIVATE feedsearch_core)
  add_test(NAME feed_interfaces COMMAND feed_tests)
endif()
